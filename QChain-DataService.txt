using System;
using System.Collections.Generic;
using System.Net.Http;
using System.Text.Json;
using System.Text.Json.Serialization;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Azure.WebJobs;
using Microsoft.Azure.WebJobs.Extensions.Http;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.Logging;

namespace QuantumBlockchain.Functions
{
    public static class FetchUsers
    {
        private static readonly HttpClient httpClient = new HttpClient();

        [FunctionName("FetchUsers")]
        public static async Task<IActionResult> Run(
            [HttpTrigger(AuthorizationLevel.Function, "get", Route = null)] HttpRequest req,
            ILogger log)
        {
            var usersRecords = new List<(int Id, string userName, string addressBC, DateTime regDate)>();

            try
            {
                string apiUrl = "http://localhost:3000/api/users"; // Replace with actual endpoint

                HttpResponseMessage response = await httpClient.GetAsync(apiUrl);
                response.EnsureSuccessStatusCode();

                string json = await response.Content.ReadAsStringAsync();

                var userResponse = JsonSerializer.Deserialize<UserApiResponse>(json, new JsonSerializerOptions
                {
                    PropertyNameCaseInsensitive = true
                });

                foreach (var user in userResponse.Users)
                {
                    usersRecords.Add((
                        int.Parse(user.Id),
                        user.Name,
                        user.Wallet,
                        DateTime.TryParse(user.RegDate, out var date) ? date : DateTime.MinValue
                    ));
                }

                // Just for demonstration, return the count
                return new OkObjectResult(new
                {
                    count = usersRecords.Count,
                    records = usersRecords
                });
            }
            catch (Exception ex)
            {
                log.LogError($"Error fetching users: {ex.Message}");
                return new StatusCodeResult(500);
            }
        }
    }

    public class UserApiResponse
    {
        [JsonPropertyName("total")]
        public int Total { get; set; }

        [JsonPropertyName("users")]
        public List<UserDto> Users { get; set; }
    }

    public class UserDto
    {
        [JsonPropertyName("id")]
        public string Id { get; set; }

        [JsonPropertyName("name")]
        public string Name { get; set; }

        [JsonPropertyName("wallet")]
        public string Wallet { get; set; }

        [JsonPropertyName("regDate")]
        public string RegDate { get; set; }
    }
}
